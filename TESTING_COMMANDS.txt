# ВНИМАНИЕ: Запускайте команды из корня проекта!

## 1. Установить зависимости
pip install -e '.[dev]'

## 2. Запустить тесты AI слоя
python -m pytest tests/test_ai_layer.py -v

## 3. Запустить все тесты
python -m pytest tests/ -v

## 4. Проверить линтером
python -m ruff check . --fix

## 5. Или использовать готовый скрипт
bash scripts/test_and_lint.sh
